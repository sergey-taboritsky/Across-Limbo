# --- A. 追加到 common/decisions/<该分类已有决议文件> 的分类块内部 ---
# 注意：本仓规范是「分类单独定义在 categories/，决议文件打开同名顶层块」，
# 因此下面两个决议必须并进已有的 MAL_legislative_council = { ... } 块，
# 不要在 categories/ 里再写 decision =，也不要在决议里写 category =。
MAL_legislative_council = {
	MAL_leg_open_parliament = {
		icon = GFX_decision_generic_factory
		fire_only_once = no
		cost = 0
		visible = { original_tag = MAL }
		complete_effect = {
			if = {
				limit = { NOT = { has_country_flag = MAL_leg_parliament_initialized } }
				MAL_leg_init_parliament_effect = yes
				set_country_flag = MAL_leg_parliament_initialized
			}
			MAL_leg_reload_seat_frames_effect = yes
			set_country_flag = MAL_leg_parliament_open
		}
		ai_will_do = { base = 0 }
	}

	MAL_leg_refresh_parliament = {
		icon = GFX_decision_generic_arrest
		fire_only_once = no
		cost = 0
		visible = {
			original_tag = MAL
			has_country_flag = MAL_leg_parliament_initialized
		}
		complete_effect = {
			MAL_leg_reload_seat_frames_effect = yes
		}
		ai_will_do = { base = 0 }
	}
}
