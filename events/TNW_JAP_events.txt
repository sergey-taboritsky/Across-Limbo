# ===== 1951 工团实验审查（联动经济改造簇 / 发展度 laws / 工团热情）=====
country_event = {
	id = JAP_event.10
	title = JAP_event.10.t
	desc = JAP_event.10.d
	picture = GFX_event_JAP_taiwannanbowan
	is_triggered_only = yes
	trigger = {
		has_country_flag = JAP_reconstruction_success_flag
	}
	immediate = {
		if = {
			limit = {
				has_country_flag = JAP_yamakawa_econ_flag
				has_country_flag = JAP_metropolitan_renovation_done_flag
				# R62: development LEVEL (not raw variable) >= 3 (or = lower-second tier)
				OR = {
					has_idea = TNW_dev_industry_3
					has_idea = TNW_dev_industry_4
					has_idea = TNW_dev_industry_5
				}
				OR = {
					has_idea = TNW_dev_welfare_3
					has_idea = TNW_dev_welfare_4
					has_idea = TNW_dev_welfare_5
				}
				has_stability > 0.5
				check_variable = { var = JAP_socialist_reform_enthusiasm_var value = 100 compare = greater_than_or_equals }
				check_variable = { var = JAP_production_management_ratio_var value = 100 compare = greater_than_or_equals }
				NOT = { has_country_flag = JAP_karl_dependence_flag }
			}
			set_country_flag = JAP_experiment_success_flag
			# R62: success clears the last two ruins layers at once
			subtract_from_variable = { var = JAP_ruins_level_var value = 2 }
			clamp_variable = { var = JAP_ruins_level_var min = 0 max = 8 }
			country_event = { id = JAP_event.200 days = 5 }
			add_stability = 0.05
		}
		else = {
			set_country_flag = JAP_experiment_failure_flag
			# R62: failure worsens ruins from tier 2 back to tier 3, to be cleared later
			set_variable = { JAP_ruins_level_var = 3 }
			# R64: failure ends both route-specific dynamic modifiers
			remove_dynamic_modifier = { modifier = JAP_noro_hegemony_dm }
			remove_dynamic_modifier = { modifier = JAP_deleonism_dm }
