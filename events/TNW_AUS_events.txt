#第三次巴尔干战争，奥地利胜利
country_event = {
	id = AUS_event.1	#事件的id
	title = {
		text = AUS_event.1.t		#事件标题文本
		trigger = {
			OR = {
				tag = AUS
				tag = YUG
				tag = ROM
			}
		}
	}
	desc = {
		text = AUS_event.1.d
		trigger = {
			OR = {
				tag = AUS
				tag = YUG
				tag = ROM
			}
		}
	}
	picture = GFX_PUM_game_2	#事件图片，注册非常简单，无序严苛遵循某种格式
	immediate = {
		hidden_effect = {
			HUN = {
				white_peace = AUS
				white_peace = YUG
				white_peace = ROM
			}
			AUS = {
				annex_country = {
					target = HUN
				}
			}
			YUG = {
				transfer_state = 45
				transfer_state = 764
			}
			ROM = {
				transfer_state = 83
				transfer_state = 76
			}
			set_global_flag = TNW_warpeace_show_gui
			AUS = {
				save_global_event_target_as = winning_country
			}
			HUN = {
				save_global_event_target_as = losing_country
			}
			add_to_variable = {
				global.TNW_warpeace_dirty = 1
			}
		}
	}
	fire_only_once = yes
	trigger = {
		AND = {
			HUN = {
				has_war_with = YUG
				has_war_with = AUS
				has_war_with = ROM
				OR = {
					has_capitulated = yes
					surrender_progress > 0.5
				}
			}
		}
	}
	#激活事件，使之自动触发的条件，满足此条件事件将可以触发
	mean_time_to_happen = {
		days = 1
	}
	hidden = yes
	option = {
		name = AUS_event.1.a
	}
}
