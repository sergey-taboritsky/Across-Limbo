	id = TUR_event.3
	title = TUR_event.3.t
	desc = TUR_event.3.d
	fire_only_once = yes
	is_triggered_only = yes
	immediate = {
		set_country_flag = TUR_straits_crisis_flag
	}
	option = {
		name = TUR_event.3.a
		add_war_support = 0.10
		add_political_power = 20
		set_country_flag = TUR_straits_defiance_flag
	}
	option = {
		name = TUR_event.3.b
		add_political_power = 50
		add_timed_idea = {
			idea = TUR_straits_humiliation
			days = 365
		}
		set_country_flag = TUR_straits_accepted_flag
	}
	option = {
		name = TUR_event.3.c
		add_political_power = 30
		set_country_flag = TUR_straits_intl_flag
		if = {
			limit = {
				has_country_flag = TUR_route_snowmoon_flag
			}
			add_stability = 0.03
		}
	}
}
