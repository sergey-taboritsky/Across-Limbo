			transfer_state = 825

				add_core_of = PRC
			}
			# 华北军整体移交：先给足装备库存，再载入华北序列并冻结编制
			add_equipment_to_stockpile = { type = infantry_equipment amount = 35000 }
			add_equipment_to_stockpile = { type = artillery_equipment amount = 500 }
			add_equipment_to_stockpile = { type = support_equipment amount = 300 }

			country_lock_all_division_template = {
				is_locked = yes
				desc = TNW_CHI_military_lock_desc
			}
		}
		# 南京方面：华北序列自编制中除名（先解锁方可删除）
		set_division_template_lock = { division_template = "甲种华北师" is_locked = no }
		set_division_template_lock = { division_template = "乙种华北师" is_locked = no }
		set_division_template_lock = { division_template = "重装华北师" is_locked = no }
		delete_unit_template_and_units = { division_template = "甲种华北师" }
		delete_unit_template_and_units = { division_template = "乙种华北师" }
		delete_unit_template_and_units = { division_template = "重装华北师" }
		hidden_effect = {
			country_event = { id = CHI_event.11 days = 5 }
		}
	}
}

			}
		}
	}
	option = {
		name = CHI_event.12.a
	}

#中原大战固定剧本 - 开战90天后仍未分出胜负：华北崩溃，无条件和平，南京吞并华北并恢复傀儡（满洲、新疆除外）
country_event = {
	id = CHI_event.13
	title = CHI_event.13.t
	desc = CHI_event.13.d
	picture = GFX_PUM_game_2
	fire_only_once = yes
	is_triggered_only = yes
	trigger = {
		has_war_with = PRC
	}
	option = {
		name = CHI_event.13.a
		hidden_effect = {
			clr_global_flag = special_peace_not_annex
		}
		CHI = {
			white_peace = PRC
			annex_country = {
				target = PRC
				transfer_troops = yes
			}
			every_owned_state = {
				limit = {
					NOT = { is_controlled_by = CHI }
				}
				transfer_state_to = CHI
			}
			every_state = {
				limit = { is_core_of = PRC }
				remove_core_of = PRC
				add_core_of = CHI
			}
			load_focus_tree = CHI_post_centralplains_diplomacy_tree
			set_country_flag = nanjing_won_cpw_tempflag
			MAL = {
				country_event = { id = MAL_dissolution.3 }
			}
			if = {
				limit = { country_exists = MEN }
				set_autonomy = { target = MEN autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = SCN }
				set_autonomy = { target = SCN autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = YUN }
				set_autonomy = { target = YUN autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = TIB }
				set_autonomy = { target = TIB autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = XSM }
				set_autonomy = { target = XSM autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = NHJ }
				set_autonomy = { target = NHJ autonomous_state = autonomy_sea_warlord_subject }
			}
			if = {
				limit = { country_exists = XKN }
				set_autonomy = { target = XKN autonomous_state = autonomy_sea_warlord_subject }
			}
		}
		PRC = {
			every_character = {
				set_nationality = CHI
			}
		}
		every_country = {
			limit = {
				is_ai = no
			}
			news_event = CHI_news_event.3
		}
	}
}
