#扎伊姆政变
country_event = {
	id = SYR_crisis.5
	title = SYR_crisis.5.t
	desc = SYR_crisis.5.d
	#picture = SYR_event5
	fire_only_once = yes
	is_triggered_only = yes
	immediate = { hidden_effect = { country_event = { id = SYR_crisis.6 days = 7 } } }
	option = {
		name = SYR_crisis.5.a
		complete_national_focus = SYR_zaim_coup
		set_politics = {
			ruling_party = Absolutism
			elections_allowed = no
		}
		set_party_name = {
			ideology = Absolutism
			name = SYR_military_junta_party
			long_name = SYR_military_junta_party_long
		}
		promote_character = SYR_Husni_al_Zaim
		set_country_flag = SYR_zaim_coup_done_flag
		mark_focus_tree_layout_dirty = yes
		remove_ideas = SYR_constitutional_debt
		add_ideas = SYR_military_junta_idea
		add_political_power = 100
		set_power_balance = {
			id = SYR_junta_balance
			left_side = SYR_attasi_bloc_bop
			right_side = SYR_zaim_army_bop
		}
		add_power_balance_value = {
			id = SYR_junta_balance
			value = 0.10
		}
	}
}
