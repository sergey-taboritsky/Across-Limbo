			name = TNW_KOR.5.a
			set_country_flag = KOR_split_done
			remove_ideas = moscow_aligned_nation
			527 = { transfer_state_to = KNR add_core_of = KNR }
			979 = { transfer_state_to = KNR add_core_of = KNR }
			982 = { transfer_state_to = KNR add_core_of = KNR }
			983 = { transfer_state_to = KNR add_core_of = KNR }

				set_country_leader = KNR_Lee_Beom_Seok

			country_event = { id = TNW_KOR.7 days = 1460 }
			KNR = { country_event = { id = TNW_KNR.1 days = 20 } }
		}
}
