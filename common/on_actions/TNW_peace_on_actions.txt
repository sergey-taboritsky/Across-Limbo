#ROOT = defeated country
#FROM = victor
on_actions = {
	on_capitulation = {
		effect = {
			#default
			if = {
				limit = {
					OR = {
						ROOT = {
							is_in_faction = no
						}
						ROOT = {
							all_allied_country = {
								has_capitulated = yes
							}
						}
					}
					###要是有这个全球flag就会直接进入谈判界面而不会自动白和###
					NOT = {
						has_global_flag = special_peace_not_annex
					}
				}
				FROM = {
					every_enemy_country = {
						transfer_navy = {
							target = FROM
						}
						limit = {
							OR = {
								original_tag = ROOT
								is_in_faction_with = ROOT
								is_subject_of = ROOT
							}
						}
						every_owned_state = {
							every_country = {
								limit = {
									NOT = {
										is_ally_with = ROOT
									}
									controls_state = PREV
								}
								transfer_state = PREV
							}
						}
						FROM = {
							annex_country = {
								target = PREV
								transfer_troops = no
							}
						}
					}
				}
			}
			if = {
				limit = {
					has_global_flag = special_peace_not_annex
				}
				clr_global_flag = special_peace_not_annex
			}
			#开启谈判结语GUI
			set_global_flag = TNW_warpeace_show_gui
			FROM = {
				save_global_event_target_as = winning_country
			}
			ROOT = {
				save_global_event_target_as = losing_country
			}
			add_to_variable = {
				global.TNW_warpeace_dirty = 1
			}
			#中原大战 - 华北胜利（南京投降：民劳接管，改为中华民国人民政府）
			if = {
				limit = {
					event_target:losing_country = {
						original_tag = CHI
					}
					event_target:winning_country = {
						original_tag = PRC
					}
				}
				clr_global_flag = special_peace_not_annex
				CHI = {
					white_peace = PRC
					annex_country = {
						target = PRC
						transfer_troops = yes
					}
					every_owned_state = {
						limit = {
							NOT = { is_controlled_by = CHI }
						}
						transfer_state_to = CHI
					}
					every_state = {
						limit = { is_core_of = PRC }
						remove_core_of = PRC
						add_core_of = CHI
					}
					load_focus_tree = CHI_post_centralplains_diplomacy_tree
					set_country_flag = CDPL_won_cpw_tempflag
					set_cosmetic_tag = CHI_PRC_gov
					set_politics = {
						ruling_party = Social_democracy
						elections_allowed = no
					}
					promote_character = CHI_Chen_Lifu
				}
				MAL = {
					country_event = { id = MAL_dissolution.1 }
				}
				PRC = {
					every_character = {
						set_nationality = CHI
					}
				}
				every_country = {
					limit = { is_ai = no }
					news_event = CHI_news_event.2
				}
			}
			#中原大战 - 南京胜利（华北投降：南京吞并华北）
			else_if = {
				limit = {
					event_target:losing_country = {
						original_tag = PRC
					}
					event_target:winning_country = {
						original_tag = CHI
					}
				}
				clr_global_flag = special_peace_not_annex
				CHI = {
					white_peace = PRC
					annex_country = {
						target = PRC
						transfer_troops = yes
					}
					every_owned_state = {
						limit = {
							NOT = { is_controlled_by = CHI }
						}
						transfer_state_to = CHI
					}
					every_state = {
						limit = { is_core_of = PRC }
						remove_core_of = PRC
						add_core_of = CHI
					}
					load_focus_tree = CHI_post_centralplains_diplomacy_tree
					set_country_flag = nanjing_won_cpw_tempflag
				}
				MAL = {
					country_event = { id = MAL_dissolution.3 }
				}
				PRC = {
					every_character = {
						set_nationality = CHI
					}
				}
				every_country = {
					limit = { is_ai = no }
					news_event = CHI_news_event.2
				}
			}
		#叙利亚内战 - 黎凡特
		else_if = {
			limit = {
				event_target:losing_country = {
					OR = {
						original_tag = SYR
						original_tag = ALP
						original_tag = LEB
						original_tag = JOR
					}
				}
				event_target:winning_country = {
					OR = {
						original_tag = SYR
						original_tag = ALP
						original_tag = LEB
						original_tag = JOR
					}
				}
			}
			event_target:losing_country = {
				every_core_state = {
					add_core_of = event_target:winning_country
					remove_core_of = event_target:losing_country
				}
			}
			event_target:winning_country = {
				annex_country = {
					tag = event_target:losing_country
				}
			}
		#澳大利亚内战 - AST vs AFF（西澳 vs 东澳）
		else_if = {
			limit = {
				OR = {
					event_target:losing_country = { original_tag = AST }
					event_target:losing_country = { original_tag = AFF }
				}
				OR = {
					event_target:winning_country = { original_tag = AST }
					event_target:winning_country = { original_tag = AFF }
				}
			}
			AST = {
				white_peace = AFF
				clr_country_flag = AST_civil_war_active
			}
			AFF = {
				white_peace = AST
			}
		}
		}
		}
	}
}
