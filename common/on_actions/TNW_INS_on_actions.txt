# INS on_actions：开局变量初始化 + 三次危机定时触发（180/360/415 天）
on_actions = {
	on_startup = {
		effect = {
			INS = {
				set_variable = { INS_goni_var = 50 }
				set_variable = { INS_marhaen_var = 50 }
				set_variable = { INS_hokokai_var = 50 }
				set_variable = { INS_sartono_var = 50 }
				set_variable = { INS_sarekat_var = 50 }
				set_variable = { INS_murba_var = 50 }
				set_variable = { INS_goni_str = 30 }
				set_variable = { INS_marhaen_str = 25 }
				set_variable = { INS_hokokai_str = 20 }
				set_variable = { INS_sartono_str = 15 }
				set_variable = { INS_sarekat_str = 10 }
				set_variable = { INS_murba_str = 10 }
				set_variable = { INS_sukarno_score = 0 }
				set_variable = { INS_yamin_score = 0 }
				country_event = { id = INS_crisis.1 days = 180 }
				country_event = { id = INS_crisis.2 days = 360 }
				country_event = { id = INS_crisis.3 days = 415 }
			}
		}
	}
}
