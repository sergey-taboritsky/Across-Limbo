		}
    }
	divide_variable = {
		var = This.SD_man_from_development_var
		value = 2000
	}








	if = {
		limit = {
			check_variable = {
				var = This.SD_man_poor_var
				value = 0
				compare = greater_than
			}
		}
		add_to_variable = {
			var = This.SD_man_rich_var
			value = This.SD_man_from_development_var
		}
		multiply_variable = {
			var = This.SD_man_from_development_var
			value = 2
		}
		subtract_from_variable = {
			var = This.SD_man_poor_var
			value = This.SD_man_from_development_var
		}
		add_to_variable = {
			var = This.SD_man_normal_var
			value = This.SD_man_from_development_var
		}
	}
	if = {
		limit = {
			check_variable = {
				var = This.SD_man_poor_var
				value = 0
				compare = equals
			}
		}
		add_to_variable = {
			var = This.SD_man_rich_var
			value = This.SD_man_from_development_var
		}
		subtract_from_variable = {
			var = This.SD_man_normal_var
			value = This.SD_man_from_development_var
		}
	}




}


SD_cal_pie_var_man = {
	#示例 原饼图比例为 50 30 20
	#将最低层饼图比例加10 此时饼图比例为 60 30 20
	clamp_variable = { #初始化当前饼图比例 限制变量范围
		var = This.SD_man_poor_var
		min = 0
		max = 100
	}
	clamp_variable = {
		var = This.SD_man_normal_var
		min = 0
		max = 100
	}
	clamp_variable = {
		var = This.SD_man_rich_var
		min = 0
		max = 100
	}

	set_variable = { #将当前饼图比例相加
		var = This.SD_man_pie_var
		value = 0
	}
	add_to_variable = {
		var = This.SD_man_pie_var
		value = This.SD_man_poor_var
	}
	add_to_variable = {
		var = This.SD_man_pie_var
		value = This.SD_man_normal_var
	}
	add_to_variable = {
		var = This.SD_man_pie_var
		value = This.SD_man_rich_var
	}
	#此时用于计算的中间临时变量为 110

	add_to_variable = { #计算出增加的饼图比列量
		var = This.SD_man_pie_var 
		value = -100
	}
	#此时用于计算的中间临时变量为 10
	divide_variable = { #均分增加的饼图比列量
		var = This.SD_man_pie_var 
		value = 3
	}
	subtract_from_variable = { #将饼图比例减去均分后的增加量
		var = This.SD_man_poor_var
		value = This.SD_man_pie_var
	}
	subtract_from_variable = { 
		var = This.SD_man_normal_var
		value = This.SD_man_pie_var
	}
	subtract_from_variable = { 
		var = This.SD_man_rich_var
		value = This.SD_man_pie_var
	}
	#此时饼图比例为 56.6 26.6 16.6
	#也就是说每次设定饼图单一比例变化只有2/3值会实际上作用于比例
	clamp_variable = { #限制变量范围
		var = This.SD_man_poor_var
		min = 0
		max = 100
	}
	clamp_variable = {
		var = This.SD_man_normal_var
		min = 0
		max = 100
	}
	clamp_variable = {
		var = This.SD_man_rich_var
		min = 0
		max = 100
	}
}

SD_state_update = {

            has_state_category = wasteland
        }
        set_state_category = pastoral
    }
    else_if = {
        limit = {
            has_state_category = enclave
        }
        set_state_category = pastoral
    }
    else_if = {
        limit = {
            has_state_category = tiny_island
        }
        set_state_category = small_island
    }
    else_if = {
        limit = {
            has_state_category = pastoral
        }
        set_state_category = rural
    }
    else_if = {

            has_state_category = small_island
        }
        set_state_category = rural

		} 
		clr_state_flag = in_civilian_economy

			ROOT = {

	}
}

SD_state_monthly_update = {
	SD_develop_economy_modify = yes
    add_to_variable = {
		var = This.SD_state_development_var
		value = This.SD_state_developing_var
    } 
    clamp_variable = {
		var = This.SD_state_development_var
		min = 0 
		max = 100
    }
    if = {
        limit = {
			NOT = {
				has_state_category = megalopolis
			}
			check_variable = {
				var = This.SD_state_development_var
				value = 100
				compare = greater_than_or_equals
			}
        }
        SD_state_update = yes
        set_variable = {
			var = This.SD_state_development_var
			value = 0
        }
        set_variable = {
			var = This.SD_state_undeveloped_var
			value = 100
        }
    }
    else_if = {
        limit = {
			has_state_category = megalopolis
			check_variable = {
				var = This.SD_state_development_var
				value = 100
				compare = greater_than_or_equals
			}
        }
        set_variable = {
			var = This.SD_state_developing_var
			value = 0
        }
    } 
    SD_build_pie_develop = yes
}
