# ============================================================
# 发展度阶梯 升级/降级 swap 效果（重构版 2026-10-10）
# 五条阶梯：education / industry / welfare / agriculture / bureaucracy
# 层级 1=最低 5=最高；到顶/到底不 swap，只给资源奖惩（与旧行为一致）
# 变量约定：
#   教育   TNW_Literacy_level_value       / TNW_Literacy_level_monthly_value       （GUI 条 1）
#   工业化 TNW_Industrial_scale_value     / TNW_Industrial_scale_monthly_value     （GUI 条 3）
#   富裕   TNW_Affluence_value            / TNW_Affluence_monthly_value            （GUI 条 2）
# 兼容说明：TNW_level_development_literacy_* 两个旧效果名保留，行为不变；
#          JAP country 文件注释及未来国策可继续按旧名调用教育阶梯。
# ============================================================

# ---------- 一、教育等级 ----------
TNW_level_development_literacy_upgrade_effect = {
	if = {
		limit = { has_idea = TNW_dev_education_1 }
		swap_ideas = {
			remove_idea = TNW_dev_education_1
			add_idea = TNW_dev_education_2
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_2 }
		swap_ideas = {
			remove_idea = TNW_dev_education_2
			add_idea = TNW_dev_education_3
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_3 }
		swap_ideas = {
			remove_idea = TNW_dev_education_3
			add_idea = TNW_dev_education_4
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_4 }
		swap_ideas = {
			remove_idea = TNW_dev_education_4
			add_idea = TNW_dev_education_5
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_5 }
		add_political_power = 50
		add_stability = 0.025
		add_war_support = 0.025
		custom_effect_tooltip = TNW_development_at_top_tt
	}
}
TNW_level_development_literacy_downgrade_effect = {
	if = {
		limit = { has_idea = TNW_dev_education_5 }
		add_political_power = -50
		add_stability = -0.025
		add_war_support = -0.025
		custom_effect_tooltip = TNW_development_at_bottom_tt
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_4 }
		swap_ideas = {
			remove_idea = TNW_dev_education_4
			add_idea = TNW_dev_education_3
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_3 }
		swap_ideas = {
			remove_idea = TNW_dev_education_3
			add_idea = TNW_dev_education_2
		}
	}
	else_if = {
		limit = { has_idea = TNW_dev_education_2 }
		swap_ideas = {
			remove_idea = TNW_dev_education_2
			add_idea = TNW_dev_education_1
		}
	}
}
