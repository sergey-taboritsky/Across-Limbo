# 苏托莫身后：双分数裁决（差距<5 = 军队出面，否则高分者就任）
INS_succession_verdict_effect = {
	set_temp_variable = { INS_score_gap = INS_sukarno_score }
	subtract_from_variable = { var = INS_score_gap value = INS_yamin_score }
	if = {
		limit = {
			check_variable = { var = INS_score_gap value = 5 compare = less_than_or_equals }
			check_variable = { var = INS_score_gap value = -5 compare = greater_than_or_equals }
		}
		set_country_flag = INS_coup_unlocked
		country_event = { id = INS_crisis.25 days = 1 }
	}
	else_if = {
		limit = { check_variable = { var = INS_score_gap value = 5 compare = less_than_or_equals } }
		set_country_flag = INS_succession_yamin
		clr_country_flag = INS_succession_sukarno
		clr_country_flag = INS_soetomo_dying
		news_event = { id = INS_news_event.3 hours = 6 }
		add_country_leader_role = {
			character = INS_Mohammad_Yamin
			promote_leader = yes
			country_leader = {
				desc = INS_Mohammad_Yamin_desc
				ideology = Guided_democracy
				traits = { }
			}
		}
	}
	else = {
		set_country_flag = INS_succession_sukarno
		clr_country_flag = INS_succession_yamin
		clr_country_flag = INS_soetomo_dying
		news_event = { id = INS_news_event.2 hours = 6 }
		add_country_leader_role = {
			character = INS_Sukarno
			promote_leader = yes
			country_leader = {
				desc = INS_Sukarno_desc
				ideology = Social_nationalism
				traits = { }
			}
		}
	}
}
