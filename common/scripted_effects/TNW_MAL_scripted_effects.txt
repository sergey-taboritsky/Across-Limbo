	# 以中执委席位最多的一派推举新主席；差值用变量相减后与 0 比较
	set_variable = { MAL_kmt_top_seats = 0 }
	set_variable = { MAL_kmt_top_faction = 1 }
	set_variable = { MAL_kmt_tmp = MAL_kmt_lodge_seats }
	subtract_from_variable = { var = MAL_kmt_tmp value = MAL_kmt_top_seats }
	if = {
		limit = {
			check_variable = { var = MAL_kmt_tmp value = 0 compare = greater_than }
			NOT = { check_variable = { var = MAL_kmt_exclude_faction value = 1 } }
		}
		set_variable = { MAL_kmt_top_seats = MAL_kmt_lodge_seats }
		set_variable = { MAL_kmt_top_faction = 1 }
